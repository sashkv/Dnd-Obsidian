	from aiogram import F  
	from aiogram import Bot, Dispatcher, types  
	from aiogram.filters import Command  
	from aiogram.types import Message  
	from aiogram.enums import ParseMode  
  
	# Вставьте сюда ваш токен  
	API_TOKEN = "7506707044:AAG7INLbEFnJsjrXrDUqxTYyPh7nkTdySm0"  
  
	# Настройка логирования  
	logging.basicConfig(level=logging.INFO)  
  
	# Создание экземпляров бота и диспетчера  
	bot = Bot(token=API_TOKEN)  
	dp = Dispatcher()  
  
	@dp.message(F.text, Command("start"))  
	async def cmd_start(message: Message):  
	    await message.answer(f"Привет, {message.from_user.full_name}, это бот повторюшка, что бы ты не отправил - вернется к тебе",parse_mode=ParseMode.HTML)  
  
	# Хэндлер для обработки всех входящих сообщений  
	@dp.message()  
	async def echo_message(message: types.Message):  
    chat_id = message.chat.id  
  
    if message.text:  
        # Отправка текстового сообщения  
        await bot.send_message(chat_id=chat_id, text=message.text)  
    elif message.sticker:  
        # Отправка стикера  
        await bot.send_sticker(chat_id=chat_id, sticker=message.sticker.file_id)  
    elif message.photo:  
        # Отправка фотографии  
        await message.answer_photo(message.photo[0].file_id)  
    elif message.voice:  
        # Отправка голосового сообщения  
        await bot.send_voice(chat_id=chat_id, voice=message.voice.file_id)  
        # Отправка гифок  
    elif message.animation:  
        await bot.send_animation(chat_id=chat_id,animation=message.animation.file_id)  
        # Отправка документов  
    elif message.document:  
        await bot.send_document(chat_id=chat_id,document=message.document.file_id)  
        # Отправка видео  
    elif message.video:  
        await bot.send_video(chat_id=chat_id,video=message.video.file_id)  
  
  
  
async def main():  
    await dp.start_polling(bot)  
  
if __name__ == "__main__":  
    asyncio.run(main())
